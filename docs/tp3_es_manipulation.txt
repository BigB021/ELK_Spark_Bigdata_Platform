# 0. Le cluster : 2 nœuds, statut green
GET _cat/nodes?v&h=name,ip,node.role,master,heap.percent
GET _cluster/health

# 1. Création de l'index avec une structure explicite
PUT ed-units-tp3
{
  "settings": { "number_of_shards": 1, "number_of_replicas": 1 },
  "mappings": {
    "properties": {
      "unit_code":   { "type": "keyword" },
      "name":        { "type": "text", "fields": { "keyword": { "type": "keyword" } } },
      "category":    { "type": "keyword" },
      "beds":        { "type": "short" },
      "pediatric":   { "type": "boolean" },
      "opened_on":   { "type": "date" },
      "description": { "type": "text" }
    }
  }
}

# 3. Vérification de la présence des documents
GET ed-units-tp3/_count
GET ed-units-tp3/_doc/U02
GET _cat/shards/ed-units-tp3?v

# 4. Recherches
# 4a. plein texte
GET ed-units-tp3/_search
{ "query": { "match": { "description": "chest pain stroke" } } }

# 4b. filtre exact + plage numérique
GET ed-units-tp3/_search
{
  "query": { "bool": {
    "filter": [ { "term": { "category": "ACUTE" } }, { "range": { "beds": { "gte": 10 } } } ]
  } }
}

# 4c. agrégation : nombre de lits par catégorie
GET ed-units-tp3/_search
{
  "size": 0,
  "aggs": { "by_category": { "terms": { "field": "category" }, "aggs": { "beds": { "sum": { "field": "beds" } } } } }
}

# 5. Mise à jour puis suppression
POST ed-units-tp3/_update/U03
{ "doc": { "beds": 10 } }
DELETE ed-units-tp3/_doc/U05
GET ed-units-tp3/_count
